function cargarMetricas() { fetch("https://smarcampus.onrender.com/asistencia/metricas") .then(res => res.json()) .then(m => { document.getElementById("metricAsistencias").textContent = m.asistencias_hoy; document.getElementById("metricFaltas").textContent = m.faltas; document.getElementById("metricTardanzas").textContent = m.tardanzas; document.getElementById("metricTotal").textContent = m.total_profesores; }) .catch(err => console.error("Error cargando métricas:", err)); } function cargarGrafica() { fetch("https://smarcampus.onrender.com/asistencia/grafica") .then(res => res.json()) .then(datos => { const ctx = document.getElementById('graficaAsistencia'); new Chart(ctx, { type: 'bar', data: { labels: datos.labels, datasets: [{ label: 'Asistencias', data: datos.values, backgroundColor: 'rgba(54, 162, 235, 0.7)' }] } }); }) .catch(err => console.error("Error cargando gráfica:", err)); } // Inicializar asistencia cargarAsistencia(); cargarMetricas(); cargarGrafica(); // Botón de filtros document.getElementById("btnFiltrar").addEventListener("click", () => { cargarAsistencia(); cargarMetricas(); cargarGrafica(); });
